CREATE TABLE IF NOT EXISTS exp3_department (
    dept_id INT PRIMARY KEY,
    department_name VARCHAR(50) NOT NULL UNIQUE,
    location VARCHAR(50) NOT NULL
);

CREATE TABLE IF NOT EXISTS exp3_project (
    project_id INT PRIMARY KEY,
    project_name VARCHAR(80) NOT NULL UNIQUE,
    budget DECIMAL(12, 2) NOT NULL
);

CREATE TABLE IF NOT EXISTS exp3_employee (
    emp_id INT PRIMARY KEY,
    full_name VARCHAR(80) NOT NULL,
    dept_id INT NOT NULL,
    designation VARCHAR(50) NOT NULL,
    salary DECIMAL(10, 2) NOT NULL,
    hire_date DATE NOT NULL,
    CONSTRAINT fk_exp3_employee_department
        FOREIGN KEY (dept_id) REFERENCES exp3_department(dept_id)
);

CREATE TABLE IF NOT EXISTS exp3_employee_project (
    emp_id INT NOT NULL,
    project_id INT NOT NULL,
    project_role VARCHAR(60) NOT NULL,
    assigned_on DATE NOT NULL,
    PRIMARY KEY (emp_id, project_id),
    CONSTRAINT fk_exp3_ep_employee
        FOREIGN KEY (emp_id) REFERENCES exp3_employee(emp_id),
    CONSTRAINT fk_exp3_ep_project
        FOREIGN KEY (project_id) REFERENCES exp3_project(project_id)
);

   Requires the Experiment 3 tables in the ByteXL MySQL workspace.

-- 1. INNER JOIN: show employees with their department names.
SELECT e.emp_id, e.full_name, d.department_name, e.salary
FROM exp3_employee AS e
INNER JOIN exp3_department AS d ON d.dept_id = e.dept_id
ORDER BY d.department_name, e.full_name;

-- 2. LEFT JOIN: include every department, even if it has no employees.
SELECT d.department_name, e.emp_id, e.full_name
FROM exp3_department AS d
LEFT JOIN exp3_employee AS e ON e.dept_id = d.dept_id
ORDER BY d.department_name, e.full_name;

-- 3. Self-join: list distinct pairs of colleagues in the same department.
SELECT d.department_name,
       e1.full_name AS employee_one,
       e2.full_name AS employee_two
FROM exp3_employee AS e1
JOIN exp3_employee AS e2
  ON e1.dept_id = e2.dept_id AND e1.emp_id < e2.emp_id
JOIN exp3_department AS d ON d.dept_id = e1.dept_id
ORDER BY d.department_name, employee_one, employee_two;

-- 4. Three-way join: employees, project assignments, and projects.
SELECT e.full_name, p.project_name, ep.project_role
FROM exp3_employee AS e
JOIN exp3_employee_project AS ep ON ep.emp_id = e.emp_id
JOIN exp3_project AS p ON p.project_id = ep.project_id
ORDER BY p.project_name, e.full_name;

-- 5. Correlated subquery: employees earning more than their own department average.
SELECT e.full_name, d.department_name, e.salary
FROM exp3_employee AS e
JOIN exp3_department AS d ON d.dept_id = e.dept_id
WHERE e.salary > (
    SELECT AVG(e2.salary)
    FROM exp3_employee AS e2
    WHERE e2.dept_id = e.dept_id
)
ORDER BY d.department_name, e.salary DESC;

-- 6. EXISTS: departments that have at least one employee earning 90,000 or more.
SELECT d.dept_id, d.department_name
FROM exp3_department AS d
WHERE EXISTS (
    SELECT 1
    FROM exp3_employee AS e
    WHERE e.dept_id = d.dept_id AND e.salary >= 90000
)
ORDER BY d.department_name;

-- 7. Simulated INTERSECT: employees assigned to both projects 101 and 106.
SELECT e.emp_id, e.full_name
FROM exp3_employee AS e
WHERE EXISTS (
    SELECT 1 FROM exp3_employee_project AS ep
    WHERE ep.emp_id = e.emp_id AND ep.project_id = 101
)
AND EXISTS (
    SELECT 1 FROM exp3_employee_project AS ep
    WHERE ep.emp_id = e.emp_id AND ep.project_id = 106
)
ORDER BY e.emp_id;

-- 8. Simulated EXCEPT: employees assigned to project 101 but not project 106.
SELECT e.emp_id, e.full_name
FROM exp3_employee AS e
WHERE EXISTS (
    SELECT 1 FROM exp3_employee_project AS ep
    WHERE ep.emp_id = e.emp_id AND ep.project_id = 101
)
AND NOT EXISTS (
    SELECT 1 FROM exp3_employee_project AS ep
    WHERE ep.emp_id = e.emp_id AND ep.project_id = 106
)
ORDER BY e.emp_id;

-- 9. Compare a query plan with EXPLAIN (MySQL syntax).
EXPLAIN
SELECT e.emp_id, e.full_name, d.department_name

WHERE e.salary > (
    SELECT AVG(e2.salary)
    FROM exp3_employee AS e2
    WHERE e2.dept_id = e.dept_id
